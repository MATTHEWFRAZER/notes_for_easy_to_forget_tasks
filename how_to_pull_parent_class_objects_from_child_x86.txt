How to pull parent class objects from child object

Example:

class X
{
private:
    int test = 3;
public:
    virtual void testx(void)
    {
        printf("hix %d\n", test);
        test++;
    }

    
};

class C : public X
{
public:
    virtual void testc(void)
    {
        printf("hic\n");
    }
    virtual void testc2(void)
    {
        printf("hi2c\n");
    }
};


class A
{
public:
    virtual void test(void)
    {
        printf("hi\n");
    }
    virtual void test2(void)
    {
        printf("hi2\n");
    }
};

class Y
{
private:
    int testingy = 0;
public:
    virtual void testy(void)
    {
        testingy++;
        printf("testy %d\n", testingy);
    }

    
};

class B : A, C, Y
{
private:
    int testing = 41;
public:
    virtual void test(void)
    {
        printf("test\n");
        test2();
        testc();
        testx();
        testy();
    }
    virtual void test2(void)
    {
        printf("%d\n", testing);
    }

    
};

Note we find a method through retrieving an object mov     rax,qword ptr [rbp+0F0h]
(see 00007ff6`4a531d0d). The vtable for B will be at 0x00. We simply add to it to
find the target method call    qword ptr [rax+8] (see 00007ff6`4a531d49) 

We find that rcx passed into the various methods is whatever vtable would be needed
for the respective method calls. Not for testx (called through the C vtable) retrieves
an object by getting object mov rax,[rbp+0xf0] and then adding to rax finds the vtable
for add rax, 8 (see 00007ff6`4a531d29 and 00007ff6`4a531d54). Similarly for Y we
see add rax, 18h (see 00007ff6`4a531d7e).

For this example, release version inlined everything, but the basic principle is the same.
